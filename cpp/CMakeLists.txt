cmake_minimum_required(VERSION 3.16)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON CACHE BOOL "Export compiler commands for clangd")
project(servo_tools LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_VIDEO "Build OpenCV video angle tool" OFF)
find_package(Threads REQUIRED)
foreach(tool bench collect install_sdk tests)
  add_executable(servo_${tool} src/${tool}.cpp)
  target_include_directories(servo_${tool} PRIVATE vendor)
  target_link_libraries(servo_${tool} PRIVATE Threads::Threads)
  if(MSVC)
    target_compile_options(servo_${tool} PRIVATE /W4 /utf-8)
  else()
    target_compile_options(servo_${tool} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
if(BUILD_VIDEO)
  find_package(OpenCV REQUIRED COMPONENTS core highgui imgproc videoio)
  add_executable(servo_video_measure src/video_measure.cpp)
  target_include_directories(servo_video_measure PRIVATE vendor ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(servo_video_measure PRIVATE ${OpenCV_LIBS})
endif()
enable_testing()
add_test(NAME core COMMAND servo_tests ${CMAKE_CURRENT_SOURCE_DIR}/../config.json)
add_executable(imu_decode_test tests/imu_decode_test.cpp)
add_test(NAME imu_decode COMMAND imu_decode_test)
add_executable(imu_bias_test tests/imu_bias_test.cpp)
target_include_directories(imu_bias_test PRIVATE vendor)
add_test(NAME imu_bias COMMAND imu_bias_test)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  add_library(bmi270_vendor STATIC vendor/bmi270/bmi2.c vendor/bmi270/bmi270.c)
  target_include_directories(bmi270_vendor PUBLIC vendor/bmi270)
  add_executable(dual_imu imu/main.cpp imu/linux_bus.cpp imu/sensors.cpp)
  target_include_directories(dual_imu PRIVATE vendor)
  target_link_libraries(dual_imu PRIVATE bmi270_vendor Threads::Threads)
  target_compile_options(dual_imu PRIVATE -Wall -Wextra -Werror)
endif()
foreach(test firmware pwm_extension e907_bench servo_service)
  add_executable(${test}_test tests/${test}_test.c)
  target_include_directories(${test}_test PRIVATE tests/stubs)
  set_target_properties(${test}_test PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON)
  if(NOT MSVC)
    target_compile_options(${test}_test PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME ${test}_simulation COMMAND ${test}_test)
endforeach()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME collector_protocol COMMAND ${Python3_EXECUTABLE}
  ${CMAKE_CURRENT_SOURCE_DIR}/tests/collect_test.py $<TARGET_FILE:servo_collect>
  ${CMAKE_CURRENT_SOURCE_DIR}/..)
