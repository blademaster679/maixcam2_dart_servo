cmake_minimum_required(VERSION 3.16)
project(servo_tools LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_VIDEO "Build OpenCV video angle tool" OFF)
find_package(Threads REQUIRED)
foreach(tool bench collect install_sdk tests)
  add_executable(servo_${tool} src/${tool}.cpp)
  target_include_directories(servo_${tool} PRIVATE vendor)
  target_link_libraries(servo_${tool} PRIVATE Threads::Threads)
  if(MSVC)
    target_compile_options(servo_${tool} PRIVATE /W4 /utf-8)
  else()
    target_compile_options(servo_${tool} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
if(BUILD_VIDEO)
  find_package(OpenCV REQUIRED COMPONENTS core highgui imgproc videoio)
  add_executable(servo_video_measure src/video_measure.cpp)
  target_include_directories(servo_video_measure PRIVATE vendor ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(servo_video_measure PRIVATE ${OpenCV_LIBS})
endif()
enable_testing()
add_test(NAME core COMMAND servo_tests ${CMAKE_CURRENT_SOURCE_DIR}/../config.json)
foreach(test firmware pwm_extension e907_bench)
  add_executable(${test}_test tests/${test}_test.c)
  target_include_directories(${test}_test PRIVATE tests/stubs)
  set_target_properties(${test}_test PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON)
  if(NOT MSVC)
    target_compile_options(${test}_test PRIVATE -Wall -Wextra -Werror)
  endif()
  add_test(NAME ${test}_simulation COMMAND ${test}_test)
endforeach()
